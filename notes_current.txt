    const len = textContent.length;
    const textNode = div.firstChild;
    const lineEnd = lineStart + len;

    // --- PHASE 1: COMPUTE LSP FORCED ZONES FOR THIS LINE ---
    let currentLSPIndex = trackedSyntax_I;
    let lspOverrideStart = -1;
    let lspOverrideEnd = -1;
    let lspType = null;

    while (currentLSPIndex < EDI_trackedSyntaxList.count_abstract) {
        EDI_trackedSyntaxList.getElementAt(currentLSPIndex);
        
        const tokenStart = INTS[fEDI_pooledTrackedSyntax_start];
        const tokenLen = INTS[fEDI_pooledTrackedSyntax_length];
        const tokenEnd = tokenStart + tokenLen;

        // Case A: This token is entirely past this line. We stop looking.
        if (tokenStart >= lineEnd) break; 

        // Case B: This token ended before this line started. Move to next token.
        if (tokenEnd <= lineStart) {
            currentLSPIndex++;
            continue;
        }

        // Case C: The token intersects this line! Compute local line relative boundaries.
        lspOverrideStart = Math.max(0, tokenStart - lineStart);
        lspOverrideEnd = Math.min(len, tokenEnd - lineStart);
        lspType = INTS[fEDI_pooledTrackedSyntax_type]; // e.g., comment code
